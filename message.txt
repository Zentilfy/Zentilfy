local function isAccentColor(c)
    local h, s, v = Color3.toHSV(c)
    return (h >= 0.55 and h <= 0.88 and s > 0.35 and v > 0.4)
        or ((h < 0.05 or h > 0.95) and s > 0.4 and v > 0.4)
        or (h > 0.08 and h < 0.20 and s > 0.4 and v > 0.5)
end

local function recolor()
    local gui = findGui(); if not gui then return end
    for _, o in ipairs(gui:GetDescendants()) do
        pcall(function()
            if o:IsA("Frame") or o:IsA("TextButton") or o:IsA("ImageButton") then
                local bg = o.BackgroundColor3
                local small = o.Size.X.Offset <= 26 and o.Size.Y.Offset <= 26
                if isAccentColor(bg) then
                    o.BackgroundColor3 = small and OUTLINE or ACCENT
                end
                if o.Parent and o.Parent:IsA("Frame") then
                    local pbg = o.Parent.BackgroundColor3
                    local parentAccent = (pbg.R > 0.85 and pbg.G > 0.85 and pbg.B > 0.85)
                    local isWhite = bg.R > 0.9 and bg.G > 0.9 and bg.B > 0.9
                    if parentAccent and isWhite and small then
                        o.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
                    end
                end
            end
            if o:IsA("ImageLabel") and isAccentColor(o.ImageColor3) then
                o.ImageColor3 = OUTLINE
            end
            if o:IsA("UIStroke") and isAccentColor(o.Color) then
                o.Color = OUTLINE
            end
            if (o:IsA("TextLabel") or o:IsA("TextButton")) then
                if isAccentColor(o.TextColor3) then
                    o.TextColor3 = ACCENT
                end
            end
        end)
    end
end

local Boosters = {

    msTarget           = 0.016,
    fpsUnlockEnabled   = false,
    fastRemotes        = false,

-- FPS unlock — pushes the renderer to run at high FPS
local function applyFpsUnlock()
    pcall(function()
        settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
    end)
    pcall(function()
        for _, v in pairs(game:GetService("Lighting"):GetChildren()) do
            if v:IsA("PostEffect") then v.Enabled = false end
        end
    end)
    pcall(function()
        game:GetService("RunService"):Set3dRenderingEnabled(true)
    end)
    -- Force a high target FPS if the client supports it
    pcall(function()
        UserSettings():GetService("UserGameSettings").MasterVolume = UserSettings():GetService("UserGameSettings").MasterVolume
    end)
end

local function startPingOverride()
    if Boosters.origPingGetValue then return end
    pcall(function()
        local serverStats = Stats.Network.ServerStatsItem["Data Ping"]
        if serverStats then
            local orig = serverStats.GetValue
            Boosters.origPingGetValue = orig
            serverStats.GetValue = newcclosure(function(self, ...)
                if Boosters.lowMsEnabled then
                    return Boosters.msTarget * 1000  -- ms
                end
                return orig(self, ...)
            end)
        end
    end)
    pcall(function()
        local clientStats = Stats.Network.ClientStatsItem and Stats.Network.ClientStatsItem["Data Ping"]
        if clientStats then
            local orig = clientStats.GetValue
            clientStats.GetValue = newcclosure(function(self, ...)
                if Boosters.lowMsEnabled then
                    return Boosters.msTarget * 1000
                end
                return orig(self, ...)
            end)
        end
    end)
end

local function stopPingOverride()
    -- Nothing needs to happen — the hook checks Boosters.lowMsEnabled at read time
end

local function setLowMs(enabled)
    Boosters.lowMsEnabled = enabled
    if enabled then
        startPingOverride()
        applyFpsUnlock()
    end
end

local function setFpsUnlock(enabled)
    Boosters.fpsUnlockEnabled = enabled
    if enabled then applyFpsUnlock() end
end

local AntiAnkle = (function()
    local state = { enabled = false, patchedMods = {}, scanDone = false }
    local PATCH_NAMES = {
        "Humillate", "AnkleCheck", "Should", "GetClosestBall",
        "InputClosestBall", "GNP", "GNB"
    }
    local function patchModule(modScript)
        if state.patchedMods[modScript] then return end
        state.patchedMods[modScript] = true
        pcall(function()
            local mod = require(modScript)
            if type(mod) ~= "table" then return end
            for _, name in ipairs(PATCH_NAMES) do
                pcall(function()
                    if mod[name] ~= nil and type(mod[name]) == "function" then
                        if name == "Should" or name == "GetClosestBall" then
                            mod[name] = function() return false end
                        elseif name == "GNP" or name == "GNB" then
                            mod[name] = function() return nil end
                        else
                            mod[name] = function() end
                        end
                    end
                end)
            end
        end)
    end
    local function scanAndPatch()
        if state.scanDone then return end
        state.scanDone = true
        local LP = Players.LocalPlayer
        if not LP then return end
        local roots = {}
        pcall(function() table.insert(roots, StarterPack) end)
        pcall(function() if LP.Character then table.insert(roots, LP.Character) end end)
        pcall(function()
            local bp = LP:FindFirstChildOfClass("Backpack")
            if bp then table.insert(roots, bp) end
        end)
        pcall(function()
            local pg = LP:FindFirstChildOfClass("PlayerGui")
            if pg then table.insert(roots, pg) end
        end)
        for _, root in ipairs(roots) do
            if root then
                pcall(function()
                    for _, obj in ipairs(root:GetDescendants()) do
                        if obj:IsA("ModuleScript") then
                            local n = string.lower(obj.Name)
                            if string.find(n, "ankle", 1, true) then patchModule(obj) end
                        end
                    end
                end)
            end
        end
    end
    local function start()
        if state.enabled then return end
        state.enabled = true
        state.scanDone = false
        state.patchedMods = {}
        task.spawn(scanAndPatch)
    end
    local function stop() state.enabled = false end
    local function rescan()
        state.scanDone = false
        state.patchedMods = {}
        task.spawn(scanAndPatch)
    end
    return { start = start, stop = stop, isRunning = function() return state.enabled end, forcePatch = rescan }
end)()

local TRAJ_RESCAN_EVERY=2 TRAJ_PREDICT_TIME=1.2 TRAJ_SIM_DT=1/60 TRAJ_MAX_BOUNCES=4
local TRAJ_RESTITUTION=0.55 TRAJ_GROUND_FRICTION=0.78 TRAJ_AIR_DRAG=0.003
local TRAJ_NUM_MARKERS=22 TRAJ_MARKER_SIZE=0.45 TRAJ_LANDING_SIZE=2.5

local trajEnabled, trajColor, trajFolder = false, WHITE, nil
local trajBalls, trajLastRescan = {}, 0
local trajLandingColor = WHITE

local function trajInitFolder()
    if not trajFolder or not trajFolder.Parent then
        trajFolder = Instance.new("Folder")
        trajFolder.Name = "Zentlify_Trajectory"
        trajFolder.Parent = workspace
    end
end
local function trajCreateLanding()
    local p = Instance.new("Part")
    p.Shape = Enum.PartType.Cylinder
    p.Size = Vector3.new(0.08, TRAJ_LANDING_SIZE, TRAJ_LANDING_SIZE)
    p.Anchored = true p.CanCollide = false p.CanQuery = false p.CanTouch = false
    p.Material = Enum.Material.Neon
    p.Color = trajLandingColor
    p.Transparency = 0.15
    p.Parent = trajFolder
    return p
end
local function trajCreateMarkers()
    local set = {}
    for i = 1, TRAJ_NUM_MARKERS do
        local m = Instance.new("Part")
        m.Shape = Enum.PartType.Ball
        m.Size = Vector3.new(TRAJ_MARKER_SIZE, TRAJ_MARKER_SIZE, TRAJ_MARKER_SIZE)
        m.Anchored = true m.CanCollide = false m.CanQuery = false m.CanTouch = false
        m.Material = Enum.Material.Neon
        m.Color = trajColor
        m.Transparency = 1
        m.Parent = trajFolder
        set[i] = m
    end
    return { markers = set, landing = trajCreateLanding() }
end
local function trajDestroySet(set)
    for _, m in ipairs(set.markers) do if m and m.Parent then m:Destroy() end end
    if set.landing and set.landing.Parent then set.landing:Destroy() end
end
local function trajRescan()
    local active = {}
    for _, b in ipairs(getBallsList()) do active[b] = true end
    for b in pairs(active) do if not trajBalls[b] then trajBalls[b] = trajCreateMarkers() end end
    for b, set in pairs(trajBalls) do
        if not b or not b.Parent or not active[b] then
            trajDestroySet(set)
            trajBalls[b] = nil
        end
    end
end
local function trajClearAll()
    for _, set in pairs(trajBalls) do trajDestroySet(set) end
    trajBalls = {}
    if trajFolder and trajFolder.Parent then trajFolder:Destroy() end
    trajFolder = nil
end
local function trajSimulate(ballPart)
    local pos = ballPart.Position
    local vel = ballPart.AssemblyLinearVelocity
    local g = workspace.Gravity
    local points = { pos }
    local landingPos = nil
    if vel.Magnitude < 0.5 then return points, pos end
    local exclude = { trajFolder }
    for p in pairs(trajBalls) do table.insert(exclude, p) end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = exclude
    local bounces = 0
    local maxSteps = math.floor(TRAJ_PREDICT_TIME / TRAJ_SIM_DT)
    for _ = 1, maxSteps do
        local stepVel = vel + Vector3.new(0, -g * TRAJ_SIM_DT, 0)
        local drag = TRAJ_AIR_DRAG * stepVel.Magnitude * TRAJ_SIM_DT
        stepVel = stepVel * math.max(0, 1 - drag)
        local stepPos = pos + stepVel * TRAJ_SIM_DT
        local rayDir = stepPos - pos
        local hit = nil
        if rayDir.Magnitude > 0 then hit = workspace:Raycast(pos, rayDir, params) end
        if hit then
            local hitPos = hit.Position
            local normal = hit.Normal.Unit
            local vn = stepVel:Dot(normal)
            if normal.Y > 0.7 and not landingPos then landingPos = hitPos end
            if vn < 0 then
                local vNormal = normal * vn
                local vTangent = stepVel - vNormal
                stepVel = vTangent * TRAJ_GROUND_FRICTION - vNormal * TRAJ_RESTITUTION
            end
            stepPos = hitPos + normal * 0.05
            bounces = bounces + 1
            table.insert(points, stepPos)
            if stepVel.Magnitude < 2.5 then break end
            if bounces > TRAJ_MAX_BOUNCES then break end
        else
            table.insert(points, stepPos)
        end
        pos = stepPos
        vel = stepVel
    end
    if not landingPos then landingPos = points[#points] end
    return points, landingPos
end
local function trajDraw(ballPart, set)
    local pts, landing = trajSimulate(ballPart)
    local step = math.max(1, math.floor(#pts / TRAJ_NUM_MARKERS))
    for i = 1, TRAJ_NUM_MARKERS do
        local idx = math.min(#pts, (i - 1) * step + 1)
        local m = set.markers[i]
        if pts[idx] and idx <= #pts then
            local frac = i / TRAJ_NUM_MARKERS
            m.Position = pts[idx]
            m.Transparency = 0.05 + 0.55 * frac
            m.Color = trajColor
            m.Size = Vector3.new(TRAJ_MARKER_SIZE, TRAJ_MARKER_SIZE, TRAJ_MARKER_SIZE) * (1 - 0.4 * frac)
        else
            m.Transparency = 1
        end
    end
    if landing and set.landing then
        set.landing.CFrame = CFrame.new(landing + Vector3.new(0, 0.04, 0)) * CFrame.Angles(0, 0, math.rad(90))
        set.landing.Size = Vector3.new(0.08, TRAJ_LANDING_SIZE, TRAJ_LANDING_SIZE)
        set.landing.Color = trajLandingColor
        set.landing.Transparency = 0.15
    end
end

local trailEnabled, trailColor = false, WHITE
local trailMap = {}

local function createTrailFor(ball)
    if trailMap[ball] then return end
    local a0 = Instance.new("Attachment"); a0.Name = "Zentlify_Trail0"; a0.Parent = ball
    local a1 = Instance.new("Attachment"); a1.Name = "Zentlify_Trail1"; a1.Position = Vector3.new(0, 0.5, 0); a1.Parent = ball
    local t = Instance.new("Trail")
    t.Name = "Zentlify_Trail"
    t.Attachment0 = a0
    t.Attachment1 = a1
    t.Lifetime = 0.5
    t.MinLength = 0
    t.FaceCamera = true
    t.LightEmission = 1
    t.LightInfluence = 0
    t.Color = ColorSequence.new(trailColor)
    t.Transparency = NumberSequence.new({
        NumberSequenceKeypoint.new(0, 0.2),
        NumberSequenceKeypoint.new(1, 1)
    })
    t.WidthScale = NumberSequence.new({
        NumberSequenceKeypoint.new(0, 1),
        NumberSequenceKeypoint.new(1, 0)
    })
    t.Parent = ball
    trailMap[ball] = { a0 = a0, a1 = a1, trail = t }
end

local function destroyTrailFor(ball)
    local data = trailMap[ball]
    if not data then return end
    pcall(function() data.trail:Destroy() end)
    pcall(function() data.a0:Destroy() end)
    pcall(function() data.a1:Destroy() end)
    trailMap[ball] = nil
end

local function refreshTrails()
    if not trailEnabled then
        for b in pairs(trailMap) do destroyTrailFor(b) end
        return
    end
    local active = {}
    for _, b in ipairs(getBallsList()) do active[b] = true createTrailFor(b) end
    for b in pairs(trailMap) do
        if not active[b] or not b.Parent then destroyTrailFor(b) end
    end
end

local function getClosestBallAny(pos)
    local cl, cd = nil, math.huge
    for _, b in ipairs(getBallsList()) do
        local d = (pos - b.Position).Magnitude
        if d < cd then cd = d cl = b end
    end
    return cl
end

local function startSmoothFollow()
    if SmoothState.connection then SmoothState.connection:Disconnect() end
    SmoothState.smoothedTarget = nil
    SmoothState.smoothedFacing = nil

    SmoothState.connection = RunService.Heartbeat:Connect(function(dt)
        if not SmoothState.guarding then return end

        local char = LP.Character
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")
        local hum  = char and char:FindFirstChildOfClass("Humanoid")
        if not hrp or not hum then return end

        if not SmoothState.lockedTarget
           or not SmoothState.lockedTarget.Parent
           or (SmoothState.lockedTarget.Position - hrp.Position).Magnitude > SmoothConfig.retargetRange then
            SmoothState.lockedTarget = getClosestBallAny(hrp.Position)
            SmoothState.smoothedTarget = nil
        end
        if not SmoothState.lockedTarget then
            hum:Move(Vector3.zero)
            return
        end

        local ball    = SmoothState.lockedTarget
        local ballPos = ball.Position
        local ballVel = ball.AssemblyLinearVelocity

        local predictedPos = ballPos + ballVel * SmoothConfig.predictionTime
        predictedPos = Vector3.new(predictedPos.X, hrp.Position.Y, predictedPos.Z)

        local toBall = predictedPos - hrp.Position
        toBall = Vector3.new(toBall.X, 0, toBall.Z)
        local dist = toBall.Magnitude
        if dist < 0.05 then return end
        local dirToBall = toBall.Unit

        local desiredPos = predictedPos - dirToBall * SmoothConfig.followDistance

        local alpha = 1 - SmoothConfig.positionSmoothing ^ (dt * 60)
        if not SmoothState.smoothedTarget then
            SmoothState.smoothedTarget = desiredPos
        else
            SmoothState.smoothedTarget = SmoothState.smoothedTarget:Lerp(desiredPos, alpha)
        end

        local toTarget = SmoothState.smoothedTarget - hrp.Position
        toTarget = Vector3.new(toTarget.X, 0, toTarget.Z)
        local toTargetMag = toTarget.Magnitude

        if toTargetMag > SmoothConfig.deadZone then
            hum:MoveTo(hrp.Position + toTarget)
        else
            hum:Move(Vector3.zero)
        end

        if SmoothConfig.faceBall and dirToBall.Magnitude > 0.1 then
            local desiredFacing = CFrame.new(hrp.Position, hrp.Position + dirToBall)
            if not SmoothState.smoothedFacing then
                SmoothState.smoothedFacing = desiredFacing
            else
                SmoothState.smoothedFacing = SmoothState.smoothedFacing:Lerp(
                    desiredFacing,
                    1 - SmoothConfig.faceSmooth ^ (dt * 60)
                )
            end
            hrp.CFrame = CFrame.new(hrp.Position)
                * (SmoothState.smoothedFacing - SmoothState.smoothedFacing.Position)
        end
    end)
end

local function stopSmoothFollow()
    if SmoothState.connection then
        SmoothState.connection:Disconnect()
        SmoothState.connection = nil
    end
    SmoothState.smoothedTarget = nil
    SmoothState.smoothedFacing = nil
    local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
    if hum then
        hum:Move(Vector3.zero)
    end
end

local function setAutoguard(s)
    SmoothState.guarding = s
    if s then
        local char = LP.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        SmoothState.lockedTarget = hrp and getClosestBallAny(hrp.Position) or nil
        startSmoothFollow()
    else
        stopSmoothFollow()
        SmoothState.lockedTarget = nil
    end
end

local pullBV, pullBall = nil, nil
local function updatePull(ball, power)
    if not ball then if pullBV then pullBV:Destroy() end pullBV, pullBall = nil, nil return end
    if pullBV and pullBall ~= ball then pullBV:Destroy() pullBV, pullBall = nil, nil end
    if not pullBV then
        pullBV = Instance.new("BodyVelocity")
        pullBV.MaxForce = Vector3.new(1e6,1e6,1e6)
        pullBV.Parent = ball
        pullBall = ball
    end
    local hrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    if pullBall and hrp then pullBV.Velocity = (hrp.Position - pullBall.Position).Unit * (power * 15) end
end
local function clearPull() if pullBV then pullBV:Destroy() end pullBV, pullBall = nil, nil end

local REACH_TOOL_NAMES = {"Shoot","Kick","Pass","Long","Tackle","Dribble","Save","Clear","GK"}
local REACH_TOOL_TYPES = {"Shoot","Pass","Long","Tackle","Dribble","Save"}

RunService.RenderStepped:Connect(function(dt)
    local char = LP.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not (hrp and hum) then return end

    local activeType = nil
    if UserInputService.TouchEnabled then
        for _, n in ipairs(REACH_TOOL_NAMES) do
            if char:FindFirstChild(n) then activeType = "Main" break end
        end
    else
        for _, n in ipairs(REACH_TOOL_TYPES) do
            if char:FindFirstChild(n) then activeType = n break end
        end
    end

    local cfgActive = activeType and ReachConfigs[activeType]
    local reachActive = cfgActive and (cfgActive.Infinite or (ReachMaster and cfgActive.Enabled))

    local boxSize = Vector3.zero
    local boxCFrame = CFrame.new(1e999,1e999,1e999)
    if reachActive then
        local sx = cfgActive.Infinite and 20000 or cfgActive.SizeX
        local sy = cfgActive.Infinite and 20000 or cfgActive.SizeY
        local sz = cfgActive.Infinite and 20000 or cfgActive.SizeZ
        boxSize = Vector3.new(sx, sy + 1, sz)
        boxCFrame = hrp.CFrame * CFrame.new(cfgActive.OffsetX, cfgActive.OffsetY, cfgActive.OffsetZ)
    end

    ReachPart.Size = boxSize
    ReachPart.CFrame = boxCFrame

    if reachActive then
        if not _G.__Zentlify_REACH_BUSY then
            _G.__Zentlify_REACH_BUSY = true
            task.spawn(function()
                pcall(function()
                    local op = OverlapParams.new()
                    op.FilterType = Enum.RaycastFilterType.Include
                    op.FilterDescendantsInstances = getBallsList()
                    local parts = workspace:GetPartsInPart(ReachPart, op)
                    if #parts > 0 then
                        table.sort(parts, function(a,b) return (a.Position - hrp.Position).Magnitude < (b.Position - hrp.Position).Magnitude end)
                        local ball = parts[1]
                        for _, ch in pairs(char:GetChildren()) do
                            if ch:IsA("BasePart") then
                                pcall(firetouchinterest, ch, ball, 0)
                                pcall(firetouchinterest, ch, ball, 1)
                            end
                        end
                    end
                end)
                task.wait(0.05)
                _G.__Zentlify_REACH_BUSY = false
            end)
        end
    else
        _G.__Zentlify_REACH_BUSY = false
    end

    if VIZ.Enabled and reachActive then
        VisualizerBox.Size = boxSize
        VisualizerBox.CFrame = boxCFrame
        VisualizerBox.Color = VIZ.Color
        VisualizerBox.Transparency = VIZ.Transparency
    else
        VisualizerBox.Size = Vector3.zero
        VisualizerBox.CFrame = CFrame.new(1e999,1e999,1e999)
        VisualizerBox.Transparency = 1
    end

    if hum then
        hum.JumpPower = MiscConfig.jumpPower
        if hum.UseJumpPower == false then
            hum.UseJumpPower = true
        end
    end

    if Toggles.CharSpeedToggle.Value and Options.CharSpeed.Value > 0 then
        local md = hum.MoveDirection
        if md.Magnitude > 0 then
            local np = hrp.CFrame + (md * dt * Options.CharSpeed.Value)
            hrp.CFrame = CFrame.new(np.Position, np.Position + hrp.CFrame.LookVector)
        end
    end

    if MiscConfig.flyEnabled then
        local fhrp = getFlyParts()
        if fhrp and flyBV and flyBG then
            local cam = workspace.CurrentCamera
            if cam then
                local vel = Vector3.zero
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then vel = vel + cam.CFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then vel = vel - cam.CFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then vel = vel - cam.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then vel = vel + cam.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then vel = vel + Vector3.new(0,1,0) end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then vel = vel - Vector3.new(0,1,0) end
                if vel.Magnitude > 0 then vel = vel.Unit * MiscConfig.flySpeed end
                flyBV.Velocity = vel
                flyBG.CFrame = cam.CFrame
            end
        end
    else
        if flyBV or flyBG then clearFly() end
    end

    if Toggles.PullVectorToggle.Value and Options.PullVectorPower.Value > 0 then
        local b = getClosestBallAny(hrp.Position)
        if b then updatePull(b, Options.PullVectorPower.Value) else clearPull() end
    else clearPull() end

    if trajEnabled then
        trajInitFolder()
        local now = tick()
        if now - trajLastRescan > TRAJ_RESCAN_EVERY then
            trajLastRescan = now
            trajRescan()
        end
        for part, set in pairs(trajBalls) do
            if part and part.Parent then
                pcall(trajDraw, part, set)
            else
                trajDestroySet(set)
                trajBalls[part] = nil
            end
        end
    end

    if MiscConfig.noFog then
        pcall(function() Lighting.FogEnd = 1e6 end)
    end

    if playerESPEnabled then
        local need = false
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP and plr.Character and not playerESPHighlights[plr] then need = true break end
        end
        if need or tick() % 1 < dt then refreshPlayerESP() end
    end
end)

Players.PlayerAdded:Connect(function() if playerESPEnabled then task.wait(0.5) refreshPlayerESP() end end)
Players.PlayerRemoving:Connect(function(plr)
    local h = playerESPHighlights[plr]
    if h then pcall(function() h:Destroy() end) playerESPHighlights[plr] = nil end
end)

    v9:Notify({Title = "Zentlify loaded", Description = "Version Number: 1.0.0 Paid", Time = 10})
end)
